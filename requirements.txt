# Pebble Engine — OPTIONAL dependencies
#
# The quiz, brief generation, and prompt download work without these.
# Install for auto-build mode (the engine calls an LLM and writes
# real website files to disk):
#
#     pip install -r requirements.txt

# --- Gemini (default provider) ---
# Google AI Ultra users get higher limits included with their subscription.
google-genai>=1.0.0

# --- Anthropic Claude (premium / second-opinion option) ---
# Switch via: PEBBLE_PROVIDER=anthropic in your .env file
anthropic>=0.40.0

# --- Post-build automation (optional) ---
# Enables `next dev` + Playwright screenshots after a full build.
# After install, also run: playwright install chromium
# Production hosts (Render) should set PEBBLE_AUTO_RUN=false — there's no
# browser to take screenshots into, and `next dev` runs locally only.
playwright>=1.40.0

# --- Cloudflare Pages publishing ---
# Pebble's /api/publish hashes static assets with BLAKE3 to compute
# Cloudflare's content-addressed upload manifest. Pure-Rust wheels are
# published on PyPI for every major OS, so no compile step on Render.
blake3>=0.4.0

# --- FastAPI + uvicorn (HTTP server, SSE streaming) ---
# Enables the FastAPI app layer with SSE progress streaming on
# /api/generate-stream. Falls back to ThreadingHTTPServer if absent.
fastapi>=0.110
uvicorn[standard]>=0.29

# --- Stripe (billing) ---
# Used by pebble/server/stripe_checkout.py (create-session), stripe_webhook.py
# (HMAC verify + subscription side-effects), billing_portal.py (customer portal
# sessions), and the one-shot bootstrap pebble/stripe_bootstrap.py.
# Tests stub the SDK entirely; this dep is only needed for the live runtime
# and the bootstrap CLI.
#
# Pinned to <16 because stripe_webhook.py explicitly documents SDK-15.x
# compatibility workarounds (search for "SDK-15" in that file). A blind
# `pip install -U` to 16.x could land breakage with no warning.
stripe>=8.0,<16.0
